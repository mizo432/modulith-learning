# 型定義設計

AG-04スプリント実行で使用されるTypeScriptの型定義を設計します。

## 1. スプリント型 (`types/sprint.ts`)

データベースの`sprints`テーブルと`sprint_backlogs`テーブルに対応します。

```typescript
// スプリント状態
export type SprintStatus = 'PLANNED' | 'ACTIVE' | 'COMPLETED' | 'CANCELLED';

// スプリント情報
export interface Sprint {
  id: number;
  productId: number;
  sprintNumber: number;
  name: string;
  goal: string | null;
  startDate: string; // ISO 8601
  endDate: string;   // ISO 8601
  status: SprintStatus;
  teamId: number | null;
  teamName: string | null;
  velocity: number | null;
  notes: string | null;
  createdAt: string;
  updatedAt: string;
}

// スプリント作成リクエスト
export interface CreateSprintRequest {
  productId: number;
  sprintNumber: number;
  name: string;
  goal: string;
  startDate: string;
  endDate: string;
  teamId: number;
}

// スプリント更新リクエスト
export interface UpdateSprintRequest {
  name?: string;
  goal?: string;
  startDate?: string;
  endDate?: string;
  status?: SprintStatus;
  teamId?: number;
  notes?: string;
}

// スプリントバックログ
export interface SprintBacklog {
  id: number;
  sprintId: number;
  capacityPoints: number;
  committedPoints: number;
  createdAt: string;
  updatedAt: string;
}

// スプリントバックログ作成リクエスト
export interface CreateSprintBacklogRequest {
  sprintId: number;
  capacityPoints: number;
  committedPoints: number;
  taskIds: number[];
}

// スプリント一覧検索条件
export interface SprintListQuery {
  productId?: number;
  status?: SprintStatus;
  keyword?: string;
  startDateFrom?: string;
  startDateTo?: string;
}

// スプリント統計情報
export interface SprintStats {
  totalSprints: number;
  activeSprints: number;
  completedSprints: number;
  cancelledSprints: number;
  averageVelocity: number | null;
}
```

## 2. タスク型 (`types/task.ts`)

データベースの`tasks`テーブルと`sprint_tasks`テーブルに対応します。

```typescript
// タスク状態
export type TaskStatus = 'TODO' | 'IN_PROGRESS' | 'DONE' | 'CANCELLED' | 'BLOCKED';

// タスク情報
export interface Task {
  id: number;
  taskCode: string;
  title: string;
  description: string | null;
  status: TaskStatus;
  estimatedHours: number | null;
  actualHours: number | null;
  progress: number; // 0-100
  assigneeId: number | null;
  assigneeName: string | null;
  sprintId: number | null;
  startDate: string | null;
  dueDate: string | null;
  completedAt: string | null;
  sourceManagementInfo: string | null;
  buildStatus: BuildStatus | null;
  notes: string | null;
  createdAt: string;
  updatedAt: string;
}

// ビルド状態
export type BuildStatus = 'SUCCESS' | 'FAILURE' | 'RUNNING';

// タスク作成リクエスト
export interface CreateTaskRequest {
  title: string;
  description?: string;
  estimatedHours?: number;
  assigneeId?: number;
  sprintId?: number;
  dueDate?: string;
}

// タスク更新リクエスト
export interface UpdateTaskRequest {
  title?: string;
  description?: string;
  status?: TaskStatus;
  estimatedHours?: number;
  actualHours?: number;
  progress?: number;
  assigneeId?: number | null;
  dueDate?: string;
  sourceManagementInfo?: string;
  notes?: string;
}

// タスク一覧検索条件
export interface TaskListQuery {
  sprintId?: number;
  status?: TaskStatus;
  assigneeId?: number;
  keyword?: string;
}
```

## 8. 共通型

```typescript
// APIレスポンスの共通型
export interface ApiResponse<T> {
  data: T;
  message?: string;
}

// ページネーション付きレスポンス
export interface PaginatedResponse<T> {
  data: T[];
  total: number;
  page: number;
  pageSize: number;
  totalPages: number;
}

// エラーレスポンス
export interface ApiError {
  status: number;
  message: string;
  details?: string;
}

// カンバンボードの列
export interface KanbanColumn {
  status: TaskStatus;
  title: string;
  tasks: Task[];
}

// カンバンボードの状態
export interface KanbanBoardState {
  columns: KanbanColumn[];
  loading: boolean;
  error: string | null;
}
```
