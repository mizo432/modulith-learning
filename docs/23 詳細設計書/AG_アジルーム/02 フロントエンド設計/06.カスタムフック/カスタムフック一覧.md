## 1. useSprint

単一のスプリント情報を取得・管理します。

```typescript
// hooks/useSprint.ts

export function useSprint(
  sprintId: number | undefined
): AsyncState<Sprint> & {
  updateSprint: (request: UpdateSprintRequest) => Promise<Sprint>;
} {
  const [state, setState] = useState<AsyncState<Sprint>>({
    data: null,
    loading: true,
    error: null,
    refetch: async () => {},
  });

  const fetchSprint = useCallback(async () => {
    if (!sprintId) return;
    setState((prev) => ({ ...prev, loading: true, error: null }));
    try {
      const data = await getSprint(sprintId);
      setState({ data, loading: false, error: null, refetch: fetchSprint });
    } catch (e) {
      setState({ data: null, loading: false, error: handleApiError(e), refetch: fetchSprint });
    }
  }, [sprintId]);

  useEffect(() => {
    fetchSprint();
  }, [fetchSprint]);

  const updateSprint = useCallback(
    async (request: UpdateSprintRequest) => {
      if (!sprintId) throw new Error('sprintId is required');
      const updated = await updateSprintApi(sprintId, request);
      setState({ data: updated, loading: false, error: null, refetch: fetchSprint });
      return updated;
    },
    [sprintId, fetchSprint]
  );

  return { ...state, refetch: fetchSprint, updateSprint };
}
```

## 3. useTasks

タスク一覧を取得・管理します。

```typescript
// hooks/useTasks.ts

export function useTasks(
  sprintId?: number
): AsyncState<PaginatedResponse<Task>> & {
  createTask: (request: CreateTaskRequest) => Promise<Task>;
  updateTask: (id: number, request: UpdateTaskRequest) => Promise<Task>;
  updateTaskStatus: (id: number, status: TaskStatus) => Promise<Task>;
  deleteTask: (id: number) => Promise<void>;
} {
  const [state, setState] = useState<AsyncState<PaginatedResponse<Task>>>({
    data: null,
    loading: true,
    error: null,
    refetch: async () => {},
  });

  const fetchTasks = useCallback(async () => {
    setState((prev) => ({ ...prev, loading: true, error: null }));
    try {
      const data = await getTasks(sprintId ? { sprintId } : {});
      setState({ data, loading: false, error: null, refetch: fetchTasks });
    } catch (e) {
      setState({ data: null, loading: false, error: handleApiError(e), refetch: fetchTasks });
    }
  }, [sprintId]);

  useEffect(() => {
    fetchTasks();
  }, [fetchTasks]);

  const updateTaskStatus = useCallback(
    async (id: number, status: TaskStatus) => {
      const updated = await updateTaskStatusApi(id, status);
      setState((prev) => ({
        ...prev,
        data: prev.data
          ? {
              ...prev.data,
              data: prev.data.data.map((t) => (t.id === id ? updated : t)),
            }
          : null,
      }));
      return updated;
    },
    []
  );

  return {
    ...state,
    refetch: fetchTasks,
    updateTaskStatus,
    // createTask, updateTask, deleteTask も同様
  };
}
```

## 6. useNotifications

通知一覧を取得・管理します。

```typescript
// hooks/useNotifications.ts

export function useNotifications(
  query: NotificationListQuery = {}
): AsyncState<PaginatedResponse<Notification>> & {
  unreadCount: number;
  markAsRead: (id: number) => Promise<void>;
  markAllAsRead: () => Promise<void>;
} {
  const [unreadCount, setUnreadCount] = useState(0);

  const fetchUnreadCount = useCallback(async () => {
    try {
      const count = await getUnreadCount();
      setUnreadCount(count);
    } catch {
      // エラー時は無視
    }
  }, []);

  // 定期的に未読数を更新
  useEffect(() => {
    fetchUnreadCount();
    const interval = setInterval(fetchUnreadCount, 60000); // 1分ごと
    return () => clearInterval(interval);
  }, [fetchUnreadCount]);

  return {
    // ...state,
    unreadCount,
    markAsRead: async (id: number) => {
      await markAsReadApi(id);
      setUnreadCount((prev) => Math.max(0, prev - 1));
    },
    markAllAsRead: async () => {
      await markAllAsReadApi();
      setUnreadCount(0);
    },
  };
}
```

export function useDailyScrum(
  sprintId: number
): AsyncState<PaginatedResponse<DailyScrum>> & {
  createScrum: (request: CreateDailyScrumRequest) => Promise<DailyScrum>;
  updateScrum: (id: number, request: UpdateDailyScrumRequest) => Promise<DailyScrum>;
  deleteScrum: (id: number) => Promise<void>;

  const [state, setState] = useState<AsyncState<PaginatedResponse<DailyScrum>>>({
    data: null,
    loading: true,
    error: null,
    refetch: async () => {},
  });

  const fetchScrums = useCallback(async () => {
    setState((prev) => ({ ...prev, loading: true, error: null }));
    try {
      const data = await getDailyScrums({ sprintId });
      setState({ data, loading: false, error: null, refetch: fetchScrums });
    } catch (e) {
      setState({ data: null, loading: false, error: handleApiError(e), refetch: fetchScrums });
    }
  }, [sprintId]);

  useEffect(() => {
    fetchScrums();
  }, [fetchScrums]);

  const createScrum = useCallback(
    async (request: CreateDailyScrumRequest) => {

    ...state,

## 9. useDragAndDrop

カンバンボードのドラッグ＆ドロップ状態を管理します。

```typescript
// hooks/useDragAndDrop.ts

export function useDragAndDrop(sprintId: number) {
  const [state, dispatch] = useReducer(kanbanReducer, {
    columns: [] as KanbanColumn[],
    loading: true,
    error: null,
  });

  const onDragEnd = useCallback(
    async (result: DropResult) => {
      if (!result.destination) return;

      const { draggableId, source, destination } = result;
      const taskId = Number(draggableId);
      const newStatus = destination.droppableId as TaskStatus;

      // オプミスティックUI更新
      dispatch({
        type: 'DRAG_END',
        payload: { taskId, destinationStatus: newStatus },
      });

      try {
        // API呼び出しで状態を永続化
        await updateTaskStatusApi(taskId, newStatus);
      } catch {
        // エラー時は元に戻す
        dispatch({
          type: 'DRAG_END',
          payload: { taskId, destinationStatus: source.droppableId as TaskStatus },
        });
      }
    },
    [sprintId]
  );

  return { state, onDragEnd };
}
```
